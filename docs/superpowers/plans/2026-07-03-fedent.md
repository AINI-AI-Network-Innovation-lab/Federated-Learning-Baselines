# FedEnt Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add FedEnt as a new baseline that preserves the current Flower evaluation flow while introducing entropy-based adaptive local learning rates and server-managed mean-field state.

**Architecture:** FedEnt will use a dedicated `FedEntStrategy` that stays FedAvg-compatible for aggregation but manages `phi1`, `phi2`, and round fit-config payloads. Client execution will be isolated in a new `training/fedent.py` module plus a small routing branch in `TorchFlowerClient`, with config, tests, and docs following existing baseline patterns.

**Tech Stack:** Python 3.10+, Flower 1.26, PyTorch 2.8, unittest, compileall

```python
def compute_fedent_learning_rate(
    *,
    parameter_vector: torch.Tensor,
    gradient_vector: torch.Tensor,
    phi1_vector: torch.Tensor,
    phi2_scalar: float,
    aggregation_weight: float,
    epsilon: float,
    max_learning_rate: float,
) -> float:
    safe_phi2 = max(float(phi2_scalar), epsilon)
    weighted_norm = float(torch.dot(parameter_vector, parameter_vector).item())
    p_i = min(1.0, max(epsilon, weighted_norm / safe_phi2))
    numerator = aggregation_weight * float(torch.dot(phi1_vector, gradient_vector).item())
    denominator = max((1.0 - aggregation_weight) * safe_phi2, epsilon)
    eta = max(0.0, numerator / denominator * (1.0 + math.log(p_i)))
    return min(max_learning_rate, eta)
```

```python
def apply_fedent_eta_decay(*, previous_eta: float | None, current_eta: float, gamma: float) -> float:
    if previous_eta is None:
        return current_eta
    return gamma * previous_eta + (1.0 - gamma) * current_eta
```

```python
def train_fedent_client(
    model: nn.Module,
    train_loader: DataLoader,
    epochs: int,
    learning_rate: float,
    device: str,
    phi1_vector: torch.Tensor,
    phi2_scalar: float,
    fedent_beta: float,
    fedent_gamma: float,
    fedent_epsilon: float,
    fedent_max_learning_rate: float,
    previous_eta: float | None,
) -> dict[str, float]:
    model.to(device)
    model.train()
    gradient_vector = estimate_full_gradient(model, train_loader, device)
    parameter_vector = flatten_model_parameters(model)
    eta_raw = compute_fedent_learning_rate(
        parameter_vector=parameter_vector,
        gradient_vector=gradient_vector,
        phi1_vector=phi1_vector,
        phi2_scalar=phi2_scalar,
        aggregation_weight=fedent_beta,
        epsilon=fedent_epsilon,
        max_learning_rate=fedent_max_learning_rate,
    )
    eta = apply_fedent_eta_decay(previous_eta=previous_eta, current_eta=eta_raw, gamma=fedent_gamma)
    optimizer = torch.optim.SGD(model.parameters(), lr=eta if eta > 0 else learning_rate)
    return run_standard_training_loop(model, train_loader, optimizer, epochs, device, eta)
```

```python
    def aggregate_fit(self, server_round, results, failures):
        aggregated_parameters, metrics = super().aggregate_fit(server_round, results, failures)
        if aggregated_parameters is None:
            return aggregated_parameters, metrics

        self._phi1 = parameters_to_ndarrays(aggregated_parameters)
        total_examples = sum(fit_res.num_examples for _, fit_res in results)
        weighted_phi2 = 0.0
        for _, fit_res in results:
            weight = fit_res.num_examples / total_examples
            weighted_phi2 += weight * float(fit_res.metrics["fedent_weight_sq_norm"])
        self._phi2 = max(weighted_phi2, self._fedent_epsilon)
        return aggregated_parameters, metrics
```
