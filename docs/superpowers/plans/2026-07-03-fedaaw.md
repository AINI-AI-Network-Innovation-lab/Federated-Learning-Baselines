# FedAAW Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add `fedaaw` as a new built-in algorithm that keeps FedAvg local training semantics while adapting server aggregation weights using per-client pre-update squared gradient norms.

**Architecture:** The implementation follows the repo's existing algorithm-builder pattern. A new `FedAAWStrategy` will extend `FedAvg` and override `aggregate_fit` to maintain gradient trackers and compute adaptive aggregation weights, while `TorchFlowerClient` routes `fedaaw` to a dedicated helper that computes the paper's client-side scalar and then trains locally with the standard objective.

**Tech Stack:** Python 3.10+, Flower 1.26, PyTorch 2.8, unittest, compileall

```python
def compute_full_batch_gradient_norm_sq(
    model: nn.Module,
    train_loader: DataLoader,
    device: str,
) -> float:
    model.train()
    model.zero_grad(set_to_none=True)
    criterion = nn.CrossEntropyLoss()
    total_loss: torch.Tensor | None = None
    total_examples = 0

    for inputs, targets in train_loader:
        inputs = inputs.to(device)
        targets = targets.to(device)
        outputs = model(inputs)
        loss = criterion(outputs, targets)
        batch_size = targets.size(0)
        weighted_loss = loss * batch_size
        total_loss = weighted_loss if total_loss is None else total_loss + weighted_loss
        total_examples += batch_size

    if total_examples == 0 or total_loss is None:
        return 0.0

    average_loss = total_loss / total_examples
    average_loss.backward()
    grad_norm_sq = 0.0
    for parameter in model.parameters():
        if parameter.grad is not None:
            grad_norm_sq += float(torch.sum(parameter.grad.detach() ** 2).item())
    model.zero_grad(set_to_none=True)
    return grad_norm_sq
```

```python
    def aggregate_fit(self, server_round, results, failures):
        if not results:
            return None, {}
        if not self.accept_failures and failures:
            return None, {}

        total_examples = sum(fit_res.num_examples for _, fit_res in results)
        if total_examples <= 0:
            return None, {}

        client_models = [
            parameters_to_ndarrays(fit_res.parameters) for _, fit_res in results
        ]
        aggregation_weights = self._adaptive_weights(server_round, results, total_examples)
        self.last_aggregation_weights = aggregation_weights
        self.global_parameters = aggregate(list(zip(client_models, aggregation_weights)))
        aggregated_parameters = ndarrays_to_parameters(self.global_parameters)
```

```python
    def _adaptive_weights(self, server_round, results, total_examples: int) -> list[float]:
        round_index = server_round - 1
        scores: list[float] = []

        for index, (client_proxy, fit_res) in enumerate(results):
            client_key = _client_key(client_proxy, index)
            grad_norm_sq = float(fit_res.metrics.get("fedaaw_grad_norm_sq", 0.0))
            tracker = self._update_tracker(client_key, grad_norm_sq, round_index)
            sample_share = fit_res.num_examples / total_examples
            score = sample_share + (self.beta / max(tracker, self.epsilon)) - self.gamma
            scores.append(score)

        weights = _softmax(scores)
        if not np.isfinite(np.asarray(weights)).all():
            return [fit_res.num_examples / total_examples for _, fit_res in results]
        return weights
```

```python
    def _update_tracker(self, client_key: str, grad_norm_sq: float, round_index: int) -> float:
        previous = self.gradient_trackers.get(client_key)
        if previous is None or round_index <= 0:
            tracker = grad_norm_sq
        else:
            tracker = ((round_index * previous) + grad_norm_sq) / (round_index + 1)
        self.gradient_trackers[client_key] = tracker
        return tracker
```
